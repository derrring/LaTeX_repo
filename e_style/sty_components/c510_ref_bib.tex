% Biblatex customizations - must use \AtEndPreamble (not begindocument hooks)
% because \renewbibmacro, \DeclareCiteCommand, \AtEveryBibitem are preamble-only.
\makeatletter
% Polyglossia registers Sanskrit as a secondary document language, but
% biblatex ships no sanskrit.lbx.  Use English bibliography strings as the
% non-lossy fallback instead of dummy definitions, unless the document or a
% style has already supplied an explicit Sanskrit mapping.
\ifcsundef{blx@lng@explicit@sanskrit}
  {\DeclareLanguageMapping{sanskrit}{english}}
  {}

\AtEndPreamble{%
    % Check if using authoryear by testing for the style file name
    \ifdefstring{\blx@bbxfile}{authoryear}{%
        % This is authoryear style - apply full customization
        \renewbibmacro*{cite}{%
            \printtext[bibhyperref]{%
                \iffieldundef{shorthand}
                {\ifthenelse{\ifnameundef{labelname}\OR\iffieldundef{labelyear}}
                    {\usebibmacro{cite:label}%
                    \setunit{\printdelim{nonameyeardelim}}}
                    {\printnames{labelname}%
                    \setunit{\printdelim{nameyeardelim}}}%
                \usebibmacro{cite:labeldate+extradate}}
        {\usebibmacro{cite:shorthand}}}}

        \DeclareCiteCommand{\cite}[\mkbibbrackets]
        {\usebibmacro{prenote}}
        {\usebibmacro{citeindex}%
        \usebibmacro{cite}}
        {\multicitedelim}
        {\usebibmacro{postnote}}
    }{%
        % Not authoryear: respect the style's native parenthesis form.
        % (Previously forced brackets via \let\mkbibparens\mkbibbrackets,
        % which collaterally overrode APA / chicago-authordate parens.
        % Numeric / alphabetic styles use brackets natively, so no-op here.)
    }

}
\makeatother

\AtEndPreamble{%
    % Other customizations
    \renewbibmacro{in:}{}
    \AtEveryBibitem{%
        \clearfield{day}%
        \clearfield{endday}%
    }
}
\endinput
