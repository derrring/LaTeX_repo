% e_class_prologue.tex
% Shared class-option prologue for the e_class family: the single source
% for the 12pt switch and the deprecated cjk option, previously
% copy-pasted verbatim into all four e_class_*.cls.
%
% The owning class must \def\e@baseclass to its base (article|book)
% before \input-ing this file:
%     \def\e@baseclass{article}
%     \input{sty_components/e_class_prologue.tex}
% =================================================================
\makeatletter

% 12pt switch. Single declaration site; read by the e_heading_scale feature.
\newif\if@e@twelvept
\@e@twelveptfalse

% Legacy [cjk] option, deprecated in the 2026 refactor: it warns and points to
% the current mechanism. CJK glyph routing is unconditional on the standard
% profile, so there is no option to disable it; a stray [nocjk] reaches the
% base class below, and LaTeX reports it as an unused option.
\DeclareOption{cjk}{\ClassWarning{\@currname}{%
    The [cjk] option is deprecated.\MessageBreak
    The standard profile already provides CJK glyph routing.\MessageBreak
    Use \string\usepackage{e_cjk_engine} for full Japanese layout%
    \MessageBreak or e_langfamily for custom per-script font families}}

% Pass 12pt to the base class and record it for e_heading_scale.
\DeclareOption{12pt}{%
    \PassOptionsToClass{\CurrentOption}{\e@baseclass}%
    \@e@twelvepttrue
}
% Pass every other option through to the base class.
\DeclareOption*{\PassOptionsToClass{\CurrentOption}{\e@baseclass}}

\ProcessOptions\relax
\LoadClass{\e@baseclass}

\makeatother
\endinput
